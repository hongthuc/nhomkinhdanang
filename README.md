        // XỬ LÝ BANNER SLIDER TỰ TRƯỢT (2 ẢNH + VUỐT CẢM ỨNG TRÊN ĐIỆN THOẠI)
        // =====================================================================
        let currentSlide = 0;
        const totalSlides = 2;
        let sliderInterval;

        function updateSlider() {
            const track = document.getElementById('sliderTrack');
            track.style.transform = `translateX(-${currentSlide * 100}%)`;
            const dots = document.querySelectorAll('.slider-dot');
            dots.forEach((dot, idx) => {
                if (idx === currentSlide) {
                    dot.className = 'slider-dot w-6 h-2 rounded-full bg-brand-accent transition-all';
                } else {
                    dot.className = 'slider-dot w-2 h-2 rounded-full bg-white/50 hover:bg-white transition-all';
                }
            });
        }

        function goToSlide(index) {
            currentSlide = (index + totalSlides) % totalSlides;
            updateSlider();
            resetSliderAutoplay();
        }

        function resetSliderAutoplay() {
            clearInterval(sliderInterval);
            sliderInterval = setInterval(() => {
                currentSlide = (currentSlide + 1) % totalSlides;
                updateSlider();
            }, 5500);
        }

        // Hỗ trợ vuốt ngang trên điện thoại cho Banner
        function initSliderTouch() {
            const slider = document.getElementById('heroSlider');
            let touchStartX = 0;
            slider.addEventListener('touchstart', (e) => {
                touchStartX = e.changedTouches[0].screenX;
            }, { passive: true });
            slider.addEventListener('touchend', (e) => {
                const touchEndX = e.changedTouches[0].screenX;
                const diff = touchStartX - touchEndX;
                if (Math.abs(diff) > 45) {
                    goToSlide(diff > 0 ? currentSlide + 1 : currentSlide - 1);
                }
            }, { passive: true });
        }
